🐍 Python API (`fourier_toolbox`) exemplo.py 
from fourier_toolbox import FourierEngine, plot_series, synthesize_wav
import numpy as np

# 1. Definir função e semiperíodo L=pi
engine = FourierEngine(
    L=np.pi,
    N=15,
    expression_or_func=lambda x: np.abs(x),
    mode='even'
)

# 2. Métricas de Parseval
parseval = engine.parseval_analysis()
print(f"Energia: {parseval['conservation_ratio']*100:.2f}%")

# 3. Exportar figura e áudio WAV
fig, ax = plot_series(engine, show_error=True)
fig.savefig("fourier.png", dpi=300)
synthesize_wav(engine, "audio.wav", base_freq=220.0)
💻 Windows PowerShell / CMD fourier-windows CLI 
# 1. Obter coeficientes analíticos e fórmula LaTeX
fourier-windows --preset abs_x -N 15 --latex

# 2. Gerar plotagem vetorial em PNG de alta resolução
fourier-windows --preset square_wave -N 25 --save-plot onda.png

# 3. Sintetizar áudio WAV na frequência fundamental
fourier-windows --preset triangle_wave -N 30 --audio som.wav

# 4. Rodar testes unitários integrados
python -m unittest discover -s tests
